(** * Classification of short exact sequences *)

(** Short exact sequences [A -> E -> B] of abelian groups are classified by pointed maps [K(B,n.+2) ->* K(A,n.+3)], for any [n : nat].  This is Christensen and Flaten, "Ext groups in homotopy type theory", Theorem 2.2.2, and the following comment.  Here we prove it for general [n] from the start. *)

Local Open Scope pointed_scope.

(** ** Applying [(K-, n.+1)] to a short exact sequence of an abelian groups gives a fiber sequence *)

Section EMFiberSequence.
  Context `{Univalence} (n : nat) {B A : AbGroup@{u}} (E : AbSES B A).

  (** The identifications [equiv_g_pi_n_em] carry [Pi n.+1] of the sequence [K(-, n.+1)] applied to [E] back to [E] itself, so that sequence is exact. *)

B, A: AbGroup
E: AbSES B A
IsExact (Tr (-1)) (fmap (Pi n.+1) (fmap (K' n.+1) (inclusion E)))
  (fmap (Pi n.+1) (fmap (K' n.+1) (projection E)))

  Defined.

  (** [K(-, n.+1)] sends short exact sequences of abelian groups to fiber sequences of Eilenberg-Mac Lane spaces.  The hypotheses of [isexact_purely_isexact_pi] are found by typeclass search: the spaces are [n]-connected and [n.+1]-truncated, [fmap (K' n.+1)] of the projection is [n]-connected since the projection is surjective, and [fmap (K' n.+1)] of the inclusion is an embedding on [Pi n.+1] since the inclusion is an embedding. *)
  #[export] Instance isexact_em_abses
    : IsExact purely (fmap (K' n.+1) (inclusion E))
        (fmap (K' n.+1) (projection E))
    := isexact_purely_isexact_pi n
        (fmap_iscomplex (K' n.+1) (iscomplex_abses E))
        isexact_pi_em_abses.

End EMFiberSequence.

(** ** The classifying map of a short exact sequence *)

(** We associate to each short exact sequence [A -> E -> B] the connecting map of the fiber sequence [K(A,n.+3) -> K(E,n.+3) -> K(B,n.+3)], expressed as a pointed map [K(B,n.+2) ->* K(A,n.+3)]. *)
Definition abses_classifying_map `{Univalence} (n : nat) {B A : AbGroup@{u}}
  (E : AbSES B A)
  : K(B, n.+2) ->* K(A, n.+3)
  := connecting_map (fmap (K' n.+3) (inclusion E))
       (fmap (K' n.+3) (projection E))
     o* pequiv_loops_em_em B n.+2.

(** ** The short exact sequence of a pointed map *)

(** Conversely, a pointed map [f : K(B,n.+2) ->* K(A,n.+3)] yields a short exact sequence [A -> Pi n.+2 (pfiber f) -> B], by rotating the fiber sequence of [f] and taking homotopy groups. *)

Section AbSESPfiber.
  Context `{Univalence} {n : nat} {B A : AbGroup@{u}}
    (f : K(B, n.+2) ->* K(A, n.+3)).

  (** The inclusion of the short exact sequence is given by the composite of the identification of [A] with [Pi n.+2 (loops K(A, n.+3))] and [Pi n.+2] applied to the connecting map [loops K(A,n+3) -> pfiber f] of the fiber sequence. *)
  Definition abses_pfiber_incl : A $-> abgroup_pi n (pfiber f)
    := fmap (Pi n.+2) (connecting_map (pfib f) f)
        $o equiv_g_pi_n_loops_em A n.+1.

  (** The projection is similarly induced by the fiber inclusion of [f]. *)
  Definition abses_pfiber_proj : abgroup_pi n (pfiber f) $-> B
    := grp_iso_inverse (equiv_g_pi_n_em B n.+1)
        $o fmap (Pi n.+2) (pfib f).

  (** The two homotopy groups neighbouring the sequence vanish: [Pi n.+2 K(A, n.+3)] lies below the connectivity of [K(A, n.+3)], and [Pi n.+3 K(B, n.+2)] lies above the truncation level of [K(B, n.+2)]. *)
  Local Instance contr_pi_em_below : Contr (Pi n.+2 K(A, n.+3))
    := contr_pi_isconnected n.+2 K(A, n.+3).

  Local Instance contr_pi_em_above : Contr (Pi n.+3 K(B, n.+2))
    := contr_pi_istrunc n.+2 K(B, n.+2).

  (** The three conditions defining a short exact sequence therefore come from the long exact sequences of the fiber sequence of [f] and of its rotation. *)

(** ** [abses_pfiber] composed with [abses_classifying_map] is the identity *)

(** In other words, the short exact sequence extracted from the classifying map of [E] is [E] itself. *)

  Context `{Univalence} (n : nat) {B A : AbGroup@{u}} (E : AbSES B A).

  (** This square arises three times below, so we state it as a local result.  The RHS is just the definition [abses_classifying_map], so the only difference is the postcomposition of the pointed identity map. *)
  Local Definition square_classifying_map
    : pequiv_pmap_idmap o* abses_classifying_map n E
      ==* connecting_map (fmap (K' n.+3) (inclusion E))
            (fmap (K' n.+3) (projection E))
          o* pequiv_loops_em_em B n.+2

  (** The fiber of the classifying map is [loops K(E,n.+3)].  Precomposing with the loop identification of [K(B,n.+2)] does not change the fiber, so it is the fiber of the connecting map, which [pequiv_pfiber_connecting_map] identifies with [loops K(E,n.+3)]. *)

(** ** The classifying map of the sequence of a delooped map *)

(** Rather than proving that the other composite is the identity, we will define an a priori different section of [abses_classifying_map].  To do this, we first show that for [psi : K(B,n.+3) ->* K(A,n.+4)], the classifying map [K(B,n.+2) ->* K(A,n.+3)] of [abses_pfiber psi] is [fmap loops psi] (with corrected endpoints) precomposed with negation on [K(B,n.+2)].  Since every map [K(B,n.+2) ->* K(A,n.+3)] can be delooped to such a [psi], this will give a section of [abses_classifying_map] in [abses_classifying_section] below. *)

Section PfiberDeloop.
  Context `{Univalence} {n : nat} {B A : AbGroup@{u}}
    (psi : K(B, n.+3) ->* K(A, n.+4)).

  (** The fiber of a map from an [n.+2]-connected type to an [n.+3]-connected type is [n.+2]-connected. *)
  Local Instance isconnected_pfiber_em : IsConnected n.+2 (pfiber psi)
    := OO_isconnected_hfiber (Tr (n.+3)%nat) (Tr (n.+2)%nat) psi pt.

  (** The fiber of [psi] is the Eilenberg-Mac Lane space of its homotopy group in degree [n.+3], by [pequiv_em_connected_truncated].  On [Pi n.+3], that identification is the inverse of [equiv_g_pi_n_em]. *)
  Local Definition fmap_pi_pequiv_em_pfiber
    (x : Pi n.+3 K(abgroup_pi n.+1 (pfiber psi), n.+3))
    : fmap (Pi n.+3) (pequiv_em_connected_truncated (pfiber psi) n.+2) x
      = (equiv_g_pi_n_em (abgroup_pi n.+1 (pfiber psi)) n.+2)^-1 x
    := fmap_pi_pequiv_em_connected_truncated (pfiber psi) n.+2 x.

  (** Through that identification, [fmap (K' n.+3)] of the projection is the fiber inclusion of [psi]. *)
  
H: Univalence
n: nat
B, A: AbGroup
psi: K( B, n.+3) ->* K( A, n.+4)
fmap (K' n.+3) (abses_pfiber_proj psi) ==*
pfib psi o* pequiv_em_connected_truncated (pfiber psi) n.+2

  
H: Univalence

    : fmap (K' n.+3) (projection F) o* fmap (K' n.+3) (component2 phi)
